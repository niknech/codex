cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
project(nutrition_bot VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BOT_FETCH_DEPS "Fetch missing JSON/date dependencies" OFF)
option(BOT_SANITIZE "Enable address/undefined sanitizers" OFF)
find_package(CURL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json CONFIG QUIET)
find_package(date CONFIG QUIET)
if(BOT_FETCH_DEPS)
  include(FetchContent)
  if(NOT nlohmann_json_FOUND)
    FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
      URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
    FetchContent_MakeAvailable(json)
  endif()
  if(NOT date_FOUND)
    set(BUILD_TZ_LIB ON CACHE BOOL "" FORCE)
    set(USE_SYSTEM_TZ_DB ON CACHE BOOL "" FORCE)
    if(WIN32)
      set(USE_SYSTEM_TZ_DB OFF CACHE BOOL "" FORCE)
    endif()
    set(ENABLE_DATE_TESTING OFF CACHE BOOL "" FORCE)
    set(MANUAL_TZ_DB ON CACHE BOOL "" FORCE)
    FetchContent_Declare(date URL https://github.com/HowardHinnant/date/archive/refs/tags/v3.0.3.tar.gz
      URL_HASH SHA256=30de45a34a2605cca33a993a9ea54e8f140f23b1caf1acf3c2fd436c42c7d942)
    FetchContent_MakeAvailable(date)
  endif()
endif()
file(READ migrations/001_initial.sql SCHEMA_SQL)
file(READ migrations/002_meal_categories.sql SCHEMA_SQL_2)
file(READ migrations/003_products.sql SCHEMA_SQL_3)
configure_file(include/schema.hpp.in generated/schema.hpp @ONLY)
add_library(bot_core src/database.cpp src/domain.cpp src/bot.cpp src/ui.cpp src/worker.cpp src/ai.cpp src/http.cpp src/chart.cpp src/config.cpp src/products.cpp src/product_flow.cpp)
target_include_directories(bot_core PUBLIC include PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
if(TARGET SQLite3::SQLite3)
  set(BOT_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(BOT_SQLITE_TARGET SQLite::SQLite3)
endif()
target_link_libraries(bot_core PUBLIC ${BOT_SQLITE_TARGET} nlohmann_json::nlohmann_json date::date-tz CURL::libcurl Threads::Threads)
target_compile_options(bot_core PRIVATE -Wall -Wextra -Wpedantic)
if(BOT_SANITIZE)
  target_compile_options(bot_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bot_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(nutrition-bot src/main.cpp)
target_link_libraries(nutrition-bot PRIVATE bot_core)
enable_testing()
add_executable(bot-tests tests/tests.cpp)
target_link_libraries(bot-tests PRIVATE bot_core)
add_test(NAME bot-tests COMMAND bot-tests)
add_executable(ui-tests tests/ui_tests.cpp)
target_include_directories(ui-tests PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(ui-tests PRIVATE bot_core)
add_test(NAME ui-tests COMMAND ui-tests)
install(TARGETS nutrition-bot RUNTIME DESTINATION bin)
